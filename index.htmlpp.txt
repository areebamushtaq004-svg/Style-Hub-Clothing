        let cart = JSON.parse(localStorage.getItem('stylehub_cart')) || [];
        let currentFilters = { category: 'all', price: 'all', size: 'all', color: 'all' };
        let currentSort = 'default';

        // ============================================
        // NAVIGATION
        // ============================================
        function navigateTo(page) {
            document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
            document.getElementById(`page-${page}`).classList.add('active');
            window.scrollTo({ top: 0, behavior: 'smooth' });
            
            if (page === 'cart') renderCart();
            if (page === 'checkout') renderCheckout();
        }

        function navigateToShop(category) {
            currentFilters.category = category;
            navigateTo('shop');
            renderShopPage();
        }

        function toggleMobileMenu() {
            document.getElementById('mobileMenu').classList.toggle('hidden');
        }

        function toggleSearch() {
            const query = prompt('Search for products:');
            if (query) {
                const results = products.filter(p => 
                    p.name.toLowerCase().includes(query.toLowerCase()) ||
                    p.category.toLowerCase().includes(query.toLowerCase())
                );
                currentFilters.category = 'all';
                navigateTo('shop');
                renderShopPage(results);
            }
        }

                              product.badge === 'Hot' ? 'badge-hot' : 'badge-new';

                <div class="product-card group cursor-pointer bg-white">

                        ${product.badge ? `<span class="absolute top-3 left-3 ${badgeClass} px-3 py-1 text-xs font-semibold uppercase tracking-wider">${product.badge}</span>` : ''}
                        <div class="absolute inset-0 bg-black/0 group-hover:bg-black/20 transition flex items-center justify-center opacity-0 group-hover:opacity-100">
                            <div class="flex gap-2">
                                <button onclick="event.stopPropagation(); viewProduct(${product.id})" class="w-11 h-11 bg-white rounded-full flex items-center justify-center hover:bg-beige-400 hover:text-white transition" title="View">

        function renderShopPage(filteredProducts = null) {
            // Update title
            const titles = {
                'all': 'All Products',
                'Men': "Men's Collection",
                'Women': "Women's Collection",
                'Kids': "Kids Collection",
                'Watches': "Watches Collection",
                'Handbags': "Handbags Collection"
            };
            document.getElementById('shopTitle').textContent = titles[currentFilters.category] || 'All Products';

            // Render category filters
            const categories = ['all', 'Men', 'Women', 'Kids', 'Watches', 'Handbags'];
            document.getElementById('categoryFilters').innerHTML = categories.map(cat => `
                <label class="flex items-center cursor-pointer hover:text-beige-500 transition">
                    <input type="radio" name="category" value="${cat}" ${currentFilters.category === cat ? 'checked' : ''} 
                           onchange="currentFilters.category='${cat}'; applyFilters()" class="mr-3 accent-beige-400">
                    ${cat === 'all' ? 'All Categories' : cat}
                </label>
            `).join('');

            // Render size filters
            const allSizes = [...new Set(products.flatMap(p => p.sizes))];
            document.getElementById('sizeFilters').innerHTML = allSizes.map(size => `
                <button onclick="toggleSizeFilter('${size}', this)" 
                        class="px-3 py-1 border border-gray-300 text-xs hover:border-brand-black hover:bg-brand-black hover:text-white transition ${currentFilters.size === size ? 'bg-brand-black text-white border-brand-black' : ''}">
                    ${size}
                </button>
            `).join('');

            // Render color filters
            const allColors = [...new Set(products.flatMap(p => p.colors))];
            document.getElementById('colorFilters').innerHTML = allColors.map(color => `
                <button onclick="toggleColorFilter('${color}', this)" 
                        class="w-8 h-8 rounded-full border-2 ${currentFilters.color === color ? 'border-brand-black scale-110' : 'border-gray-300'} transition"
                        style="background-color: ${color}" title="${color}"></button>
            `).join('');

            // Apply filters
            applyFilters(filteredProducts);
        }

        function toggleSizeFilter(size, btn) {
            currentFilters.size = currentFilters.size === size ? 'all' : size;
            renderShopPage();
        }

        function toggleColorFilter(color, btn) {
            currentFilters.color = currentFilters.color === color ? 'all' : color;
            renderShopPage();
        }

        function applyFilters(preFiltered = null) {
            let filtered = preFiltered || [...products];

            // Category filter
            if (currentFilters.category !== 'all') {
                filtered = filtered.filter(p => p.category === currentFilters.category);
            }

            // Price filter
            const priceVal = document.querySelector('input[name="price"]:checked')?.value || 'all';
            if (priceVal !== 'all') {
                if (priceVal === '0-50') filtered = filtered.filter(p => p.price < 50);
                else if (priceVal === '50-100') filtered = filtered.filter(p => p.price >= 50 && p.price <= 100);
                else if (priceVal === '100-200') filtered = filtered.filter(p => p.price >= 100 && p.price <= 200);
                else if (priceVal === '200+') filtered = filtered.filter(p => p.price >= 200);
            }

            // Size filter
            if (currentFilters.size !== 'all') {
                filtered = filtered.filter(p => p.sizes.includes(currentFilters.size));
            }

            // Color filter
            if (currentFilters.color !== 'all') {
                filtered = filtered.filter(p => p.colors.includes(currentFilters.color));
            }

            // Sort
            const sort = document.getElementById('sortSelect').value;
            if (sort === 'price-low') filtered.sort((a, b) => a.price - b.price);
            else if (sort === 'price-high') filtered.sort((a, b) => b.price - a.price);
            else if (sort === 'name') filtered.sort((a, b) => a.name.localeCompare(b.name));

            // Render
            document.getElementById('shopGrid').innerHTML = filtered.length > 0 
                ? filtered.map(createProductCard).join('')
                : '<p class="col-span-full text-center py-12 text-brand-gray">No products found matching your filters.</p>';
            document.getElementById('productCount').textContent = `${filtered.length} product${filtered.length !== 1 ? 's' : ''}`;
        }

        function resetFilters() {
            currentFilters = { category: 'all', price: 'all', size: 'all', color: 'all' };
            document.querySelector('input[name="price"][value="all"]').checked = true;
            document.getElementById('sortSelect').value = 'default';
            renderShopPage();
        }

            btn.parentElement.querySelectorAll('button').forEach(b => {
                b.classList.remove('border-brand-black', 'bg-brand-black', 'text-white');
                b.classList.add('border-gray-300');

            btn.parentElement.querySelectorAll('button').forEach(b => {
                b.classList.remove('border-brand-black', 'scale-110');
                b.classList.add('border-gray-300');
            });
            btn.classList.remove('border-gray-300');
            btn.classList.add('border-brand-black', 'scale-110');
        }

        function changeQty(delta) {
            const input = document.getElementById('productQty');
            const newVal = Math.max(1, parseInt(input.value) + delta);
            input.value = newVal;
        }

        // ============================================
        // CART FUNCTIONALITY
        // ============================================
        function addToCart(productId, qty = 1) {
            const product = products.find(p => p.id === productId);
            const existing = cart.find(item => item.id === productId);
            
            if (existing) {
                existing.quantity += qty;
            } else {
                cart.push({
                    id: product.id,
                    name: product.name,
                    price: product.price,
                    image: product.images[0],
                    category: product.category,
                    quantity: qty
                });
            }
            
            saveCart();
            updateCartBadge();
            showToast(`${product.name} added to cart!`);
        }

        function addToCartWithQty(productId) {
            const qty = parseInt(document.getElementById('productQty').value) || 1;
            addToCart(productId, qty);
        }

        function buyNow(productId) {
            addToCartWithQty(productId);
            navigateTo('checkout');
        }

        function updateCartQty(index, delta) {
            cart[index].quantity += delta;
            if (cart[index].quantity <= 0) cart.splice(index, 1);
            saveCart();
            updateCartBadge();
            renderCart();
        }

        function removeFromCart(index) {
            cart.splice(index, 1);
            saveCart();
            updateCartBadge();
            renderCart();
        }

        function saveCart() {
            localStorage.setItem('stylehub_cart', JSON.stringify(cart));
        }

        function updateCartBadge() {
            const count = cart.reduce((sum, item) => sum + item.quantity, 0);
            document.getElementById('cartBadge').textContent = count;
        }

        function placeOrder(e) {
            e.preventDefault();
            if (cart.length === 0) {
                showToast('Your cart is empty!');
                return;
            }
            const total = cart.reduce((sum, item) => sum + item.price * item.quantity, 0);
            alert(`Order placed successfully!\n\nTotal: $${total.toFixed(2)}\n\nThank you for shopping with StyleHub!`);
            cart = [];
            saveCart();
            updateCartBadge();
            navigateTo('home');
        }
